# Mind Map Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `mind-map` block type whose IR is a nested tree and whose figure is a horizontal dendrogram, so containment is visible at a glance.

**Architecture:** A new `src/figures/layoutTree.ts` runs a two-pass dendrogram — post-order to compute subtree extents, pre-order to place nodes so each parent is centred over its own children — and emits the existing `Scene` IR, which `sceneToSvg` already serialises. Connectors are one orthogonal trunk per parent (a multi-segment `line-path`), not one edge per child, so they cannot self-overlap. The IR is a nested tree rather than nodes-plus-edges so cycles, orphans and multi-parent nodes are unrepresentable. `auto` presentation degrades to a nested `<ul>` on a node count **and** a measured scene width, never on count alone.

**Tech Stack:** TypeScript, zod, Temml (MathML), vitest, Vite. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-04-mind-map-design.md`

```ts
      children: [{ label: "In-memory" }, { label: "On-disk" }],

    { label: "Eviction", children: [{ label: "LRU" }, { label: "LFU" }] },

function boxes(scene: Scene): NodeBoxEl[] {
  return scene.elements.filter((e): e is NodeBoxEl => e.kind === "node-box")
}

/** Rebuilds parent→child from the emitted Scene. Trunk paths encode which
 *  children hang off which parent by their vertical span, so the invariant is
 *  checked against the OUTPUT rather than against internal state — a layout
 *  that computes the right number and serialises the wrong box fails here. */
function childrenOf(scene: Scene, parent: NodeBoxEl): NodeBoxEl[] {
  const all = boxes(scene)
  const out: NodeBoxEl[] = []
  for (const b of all) {
    if (b.x <= parent.x + parent.w) continue
    // A child is the nearest node to the right on the same row band.
    const dy = Math.abs(b.y + b.h / 2 - (parent.y + parent.h / 2))
    if (dy < b.h) out.push(b)
  }
  return out.sort((a, b) => a.y - b.y)
}

function centre(b: NodeBoxEl): number { return b.y + b.h / 2 }

describe("layoutTree — the centring invariant", () => {
  it("centres every parent over its own children's span", () => {
    const { scene } = layoutTree(CACHE_TREE)
    const all = boxes(scene)
    for (const p of all) {
      const kids = childrenOf(scene, p)
      if (kids.length === 0) continue
```

```ts
import type { LinePathEl, NodeBoxEl, NodeShape, Scene } from "./scene"

/** One node of a containment tree. Deliberately the same shape the block's
 *  IR uses, so the layout has no translation layer to get out of sync. */
export interface TreeNodeSpec {
  label: string
  children?: TreeNodeSpec[]
}

export const TREE_PAD = 16
export const TREE_H_GAP = 44
export const TREE_V_GAP = 18
export const TREE_NODE_W_MAX = 176
export const TREE_NODE_W_MIN = 64
export const TREE_NODE_H = 44
const CHAR_W = 7.5 // per-char estimate at 13px, same basis as layoutFlow
const NODE_PAD = 24

/** Widest scene the figure form may occupy (spec §4.4). Derived, not chosen:
 *  a scene of width W scales to the column, so an 11px label renders at
 *  11 x (column / W); holding that at a readable 8px in a ~600px column puts
 *  the ceiling at 825. Past it the figure is not a wide diagram, it is a
 *  shrunken one. */
export const TREE_MAX_SCENE_W = 825

interface Measured {
  label: string
  depth: number
  w: number
  h: number
  /** Total vertical band this node's whole subtree occupies. */
  subtreeH: number
  kids: Measured[]
  x: number
  y: number
}

function n(v: number): string {
  return String(Math.round(v * 100) / 100)
}

function wrap(label: string, avail: number): string[] {
  const maxChars = Math.max(8, Math.floor((avail - NODE_PAD) / CHAR_W))
  if (label.length <= maxChars) return [label]
  const words = label.split(/\s+/)
  const lines: string[] = []
  let line = ""
  for (const word of words) {
    const next = line ? `${line} ${word}` : word
    if (next.length > maxChars && line) { lines.push(line); line = word } else { line = next }
  }
  if (line) lines.push(line)
  return lines.length > 0 ? lines : [label.slice(0, maxChars)]
}

/** Pass 1 — post-order. Subtree extent only; nothing is placed yet. */
function measure(spec: TreeNodeSpec, depth: number): Measured {
  const lines = wrap(spec.label, TREE_NODE_W_MAX)
  const longest = Math.max(...lines.map(l => l.length))
  const w = Math.min(TREE_NODE_W_MAX, Math.max(TREE_NODE_W_MIN, Math.round(longest * CHAR_W) + NODE_PAD))
  const h = lines.length === 1 ? TREE_NODE_H : lines.length * 18 + 20
  const kids = (spec.children ?? []).map(c => measure(c, depth + 1))
  const subtreeH = kids.length === 0
    ? h
    : kids.reduce((s, k) => s + k.subtreeH, 0) + (kids.length - 1) * TREE_V_GAP
  return { label: spec.label, depth, w, h, subtreeH, kids, x: 0, y: 0 }
}

function eachNode(m: Measured, fn: (n: Measured) => void): void {
  fn(m)
  for (const k of m.kids) eachNode(k, fn)
}

function flatten(m: Measured): Measured[] {
  const out: Measured[] = []
  eachNode(m, n => out.push(n))
  return out
}

/**
 * A horizontal dendrogram. Pass 1 measures subtree extents bottom-up; pass 2
 * seats each child subtree inside its parent's band with the PARENT CENTRED
 * OVER IT. That centring is the whole reason this is not `layoutFlow`, which
 * ranks children and left-aligns them — measured at ~162px off on the spec's
 * own example tree.
 *
 * Columns hug their own widest label, so a tree of short names stays narrow
 * instead of inheriting the width of one long one.
 */
export function layoutTree(root: TreeNodeSpec): { scene: Scene } {
  const m = measure(root, 0)

  const colW: number[] = []
  for (const node of flatten(m)) {
    colW[node.depth] = Math.max(colW[node.depth] ?? 0, node.w)
  }
  const colX: number[] = []
  let acc = TREE_PAD
  for (const w of colW) { colX.push(acc); acc += w + TREE_H_GAP }

  // Pass 2 — pre-order placement.
  const place = (node: Measured, top: number): void => {
    node.x = colX[node.depth]!
    node.y = top + (node.subtreeH - node.h) / 2
    let cursor = top
    for (const k of node.kids) {
      place(k, cursor)
      cursor += k.subtreeH + TREE_V_GAP
    }
  }
  place(m, TREE_PAD)

  const all = flatten(m)
  const width = TREE_PAD * 2 + colW.reduce((s, w) => s + w, 0) + TREE_H_GAP * (colW.length - 1)
  const height = TREE_PAD * 2 + m.subtreeH

  const ids = new Map<Measured, string>()
  all.forEach((node, i) => ids.set(node, `n${i}`))

  // Shape carries depth, so hierarchy reads without the connectors: this is
  // the same vocabulary `diagram` already uses, not a new one.
  const shapeFor = (depth: number): NodeShape =>
    depth === 0 ? "pill" : depth === 1 ? "box" : "note"

  const elements: Scene["elements"] = []
  for (const node of all) {
    const lines = wrap(node.label, node.w)
    const box: NodeBoxEl = {
      kind: "node-box", id: ids.get(node)!, x: node.x, y: node.y,
      w: node.w, h: node.h, shape: shapeFor(node.depth), label: node.label,
    }
    if (lines.length > 1) box.lines = lines
    elements.push(box)
  }

  // Connectors: ONE trunk per parent. An elbow from the parent's centre to
  // each child's centre would stack every vertical segment on one x and
  // overlap — the standard hand-rolled-tree mistake. A single vertical trunk
  // at the children's column, reached by one horizontal stub from the parent,
  // cannot self-overlap because the children are contiguous in y by
  // construction (pass 1). No arrowhead: containment is not directed.
  for (const node of all) {
    if (node.kids.length === 0) continue
    const exitX = node.x + node.w
    const trunkX = node.kids[0]!.x
    const pcy = node.y + node.h / 2
    const first = node.kids[0]!.y + node.kids[0]!.h / 2
    const last = node.kids[node.kids.length - 1]!.y + node.kids[node.kids.length - 1]!.h / 2
    const d = node.kids.length === 1
      ? `M ${n(exitX)} ${n(pcy)} L ${n(trunkX)} ${n(first)}`
      : `M ${n(exitX)} ${n(pcy)} L ${n(trunkX)} ${n(pcy)} L ${n(trunkX)} ${n(first)} L ${n(trunkX)} ${n(last)}`
    const trunk: LinePathEl = { kind: "line-path", d }
    elements.push(trunk)
  }

  return { scene: { width, height, elements } }
}

export function mindMapTreeStats(root: TreeNodeSpec): { nodes: number; depth: number } {
  const m = measure(root, 0)
  let nodes = 0
  let depth = 0
  eachNode(m, x => { nodes++; depth = Math.max(depth, x.depth) })
  return { nodes, depth: depth + 1 }
}

export function mindMapFigureWidth(root: TreeNodeSpec): number {
  return layoutTree(root).scene.width
}

export function mindMapFigureFits(root: TreeNodeSpec): boolean {
  return mindMapFigureWidth(root) <= TREE_MAX_SCENE_W
}
```

### Task 2: The schema, explicit to depth 12

Independent of Task 1 — different files, no shared types. Done second because it is the smallest self-contained deliverable and it carries the crash guard.

**Why the IR is nested rather than `nodes` + `edges`, so nobody "simplifies" it later.** `diagram` takes a node list and an edge list, and a containment tree *could* be expressed as one — so the flattening looks like a saving until you count what it costs. A model authors nested structure far more reliably than it invents ids and parent→child edges, and `diagram`'s edge list let the flow-cycle cycle spend a session on back-edge classification precisely because a model *can* express a cycle there. A nested tree cannot: no cycles, no orphans, no nodes with two parents. Those shapes become unrepresentable rather than detected — the invalid case resolved structurally instead of defensively. Containment *is* nesting, so the edges would be a lossy encoding the model has to get right on every call.

```ts
 * legibility policy: see MIND_MAP_MAX_DEPTH in primitives/mind-map.ts, which
 * is a different number doing a different job.
 *

 * A recursive schema throws RangeError: Maximum call stack size exceeded on a
 * 20,000-deep tree, and safeParse does NOT catch it — salvage.ts calls
 * safeParse with no try/catch, so it would escape the render pass as an
 * uncaught exception. Measured 2026-10-04. Explicit nesting makes that
 * unrepresentable.

/** Per-level breadth cap. */
export const MIND_MAP_MAX_CHILDREN = 8

  z.array(inner).min(1).max(MIND_MAP_MAX_CHILDREN).optional()

const leaf = z.object({ label: treeLabel })
const d1 = z.object({ label: treeLabel, children: treeChildren(leaf) })
const d2 = z.object({ label: treeLabel, children: treeChildren(d1) })
const d3 = z.object({ label: treeLabel, children: treeChildren(d2) })
const d4 = z.object({ label: treeLabel, children: treeChildren(d3) })
const d5 = z.object({ label: treeLabel, children: treeChildren(d4) })
const d6 = z.object({ label: treeLabel, children: treeChildren(d5) })
const d7 = z.object({ label: treeLabel, children: treeChildren(d6) })
const d8 = z.object({ label: treeLabel, children: treeChildren(d7) })
const d9 = z.object({ label: treeLabel, children: treeChildren(d8) })
const d10 = z.object({ label: treeLabel, children: treeChildren(d9) })
const d11 = z.object({ label: treeLabel, children: treeChildren(d10) })
```

```ts
import { registerPrimitive, type RenderContext } from "../core/registry"
import { mindMapBlockSchema, type MindMapBlock, type MindMapNode } from "../core/schema"
import { renderInline } from "../core/inline"
import { layoutTree, mindMapTreeStats, mindMapFigureFits, TREE_MAX_SCENE_W, type TreeNodeSpec } from "../figures/layoutTree"
import { sceneToSvg } from "../figures/svg"
import { hashSeed } from "../figures/seed"

/** Legibility policy — how deep a readable tree is. Deliberately NOT the
 *  schema's crash guard (MIND_MAP_SCHEMA_DEPTH = 12): a 6-deep tree is valid
 *  but unreadable, so it degrades to text and says so, rather than being
 *  dropped as an invalid block. */
export const MIND_MAP_MAX_DEPTH = 5
/** Crude first guard, subordinate to the measured width check. */
export const MIND_MAP_MAX_NODES = 24

/** Why the list form was chosen, when it was a DEGRADATION rather than a
 *  request. `reason` is absent for an explicit `presentation`, so the caller
 *  cannot report a degradation the author actually asked for. */
type Presentation = { form: "text" | "figure"; reason?: string }

/**
 * Figure or nested list?
 *
 * Degrades on THREE independent conditions, and the width check is the one
 * that matters most — count alone let a 3-event timeline with 40-character
 * names through once, and the figure silently overlapped its own labels. A
 * tree of 24 one-word labels is very wide; a tree of 6 long ones is very
 * deep. `mindMapFigureFits` measures the laid-out scene against the 825px
 * ceiling, the same arithmetic layoutTimeline uses.
 *
 * Returning the REASON rather than a bare "text" matters: without it an
 * explicit `presentation: "text"` is indistinguishable from a degradation,
 * and the host gets a warning blaming the width for a choice the author made.
 *
 * The list is the better outcome, not a consolation prize: it is also the
 * degradation target, the accessible form, and the no-SVG fallback, so those
 * three cannot drift apart.
 */
function resolvePresentation(block: MindMapBlock): Presentation {
  if (block.presentation === "text") return { form: "text" }
  if (block.presentation === "figure") return { form: "figure" } // deliberate override
  const stats = mindMapTreeStats(block.root as TreeNodeSpec)
  if (stats.nodes > MIND_MAP_MAX_NODES) {
    return { form: "text", reason: `${stats.nodes} nodes exceeds ${MIND_MAP_MAX_NODES}` }
  }
  if (stats.depth > MIND_MAP_MAX_DEPTH) {
    return { form: "text", reason: `depth ${stats.depth} exceeds ${MIND_MAP_MAX_DEPTH}` }
  }
  if (!mindMapFigureFits(block.root as TreeNodeSpec)) {
    return { form: "text", reason: `scene is wider than ${TREE_MAX_SCENE_W}px` }
  }
  return { form: "figure" }
}

/** "Cache contains Browser cache, Shared cache; Shared cache contains
 *  In-memory, On-disk." The figure carries no text a screen reader can use
 *  otherwise, and the SVG has no accessible structure of its own. */
function ariaFor(node: MindMapNode): string {
  const kids = node.children ?? []
  if (kids.length === 0) return ""
  const parts = kids.map(k => (k.children?.length ? ariaFor(k) : renderInline(k.label, undefined)))
  return `${renderInline(node.label, undefined)} contains ${parts.join(", ")}`
}

function listHtml(node: MindMapNode): string {
  const kids = node.children ?? []
  const item = `<li>${renderInline(node.label, undefined)}`
  return kids.length === 0
    ? `${item}</li>`
    : `${item}<ul>${kids.map(listHtml).join("")}</ul></li>`
}

export function renderMindMap(block: MindMapBlock, ctx: RenderContext): string {
  const { form, reason } = resolvePresentation(block)
  if (form === "text") {
    if (reason) ctx.addWarning(`[mind-map] ${reason} — rendered as a nested list`)
    return `<div class="exp-block exp-mind-map exp-mind-map-text"><ul class="exp-mind-map-list">${listHtml(block.root)}</ul></div>`
  }
  const { scene } = layoutTree(block.root as TreeNodeSpec)
  const aria = ariaFor(block.root) || renderInline(block.root.label, undefined)
  const svg = sceneToSvg(scene, { ariaLabel: aria, sketch: ctx.sketch, seed: block.id ?? hashSeed(aria) })
  return `<div class="exp-block exp-mind-map exp-mind-map-figure">${svg}</div>`
}

registerPrimitive("mind-map", { schema: mindMapBlockSchema, render: renderMindMap })
```
