# Demo Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A zero-dependency static demo (`npm run demo`) rendering one education and one data-driven explanation in both themes, served from built `dist/` artifacts.

**Architecture:** Plain-ESM IR fixtures in `demo/examples.js` (schema-checked by vitest), rendered by the built core's `renderExplanation(ir, { theme: "none" })` into one static page; theme switching toggles `exp-theme-light`/`exp-theme-dark` classes on page sections — demonstrating the token architecture (one render, two themes).

**Tech Stack:** Node `http` (no new deps), existing vitest suite, existing tsup build.

**Spec:** `docs/superpowers/specs/2026-09-27-demo-page-design.md`

## Global Constraints

- No new runtime or dev dependencies (spec: zero-dependency demo).
- Serve from `dist/` only, never `src/` (spec decision).
- Explanations render with `theme: "none"`; themes come from page-section classes (spec decision).
- Both fixtures must parse against `explanationSchema` (spec requirement).
- Education fixture covers: `concept`, `worked-example`, `derivation`, `callout`, `stack`. Data fixture covers: `data-callout`, `comparison`, `cause-effect`, `timeline`, `callout`.
- Port 4173, overridable via `PORT`; inline 3-line toggle script only.

```ts
import { describe, expect, it } from "vitest"
import { explanationSchema } from "../src/core/schema"
import { educationIr, dataIr } from "../demo/examples.js"

function collectTypes(ir: { blocks: unknown[] }): Set<string> {
  const types = new Set<string>()
  for (const raw of ir.blocks) {
    const b = raw as { type?: string; blocks?: unknown[] }
    if (typeof b.type !== "string") continue
    if (b.type === "stack" && Array.isArray(b.blocks)) {
      types.add("stack")
      for (const child of b.blocks) {
        const c = child as { type?: string }
        if (typeof c.type === "string") types.add(c.type)
      }
    } else {
      types.add(b.type)
    }
  }
  return types
}

describe("demo examples", () => {
  it("education fixture is schema-valid", () => {
    expect(explanationSchema.safeParse(educationIr).success).toBe(true)
  })

  it("data fixture is schema-valid", () => {
    expect(explanationSchema.safeParse(dataIr).success).toBe(true)
  })

  it("education fixture covers concept, worked-example, derivation, callout, stack", () => {
    const types = collectTypes(educationIr)
    for (const t of ["concept", "worked-example", "derivation", "callout", "stack"]) {
      expect(types.has(t)).toBe(true)
    }
  })

  it("data fixture covers data-callout, comparison, cause-effect, timeline, callout", () => {
    const types = collectTypes(dataIr)
    for (const t of ["data-callout", "comparison", "cause-effect", "timeline", "callout"]) {
      expect(types.has(t)).toBe(true)
    }
  })
})
```

- [ ] **Step 1: Write `demo/server.mjs`**

```js
// Zero-dependency demo server. Serves the built package (dist/) — doubles as a
// smoke test of the published artifact shape. `npm run demo` builds first.
import { createServer } from "node:http"
import { readFile } from "node:fs/promises"
import { renderExplanation } from "../dist/index.js"
import { educationIr, dataIr } from "./examples.js"

const port = Number(process.env.PORT ?? 4173)

function section(id, theme, ir) {
  const { html } = renderExplanation(ir, { theme: "none" })
  return `<section id="${id}" class="exp exp-theme-${theme}">${html}</section>`
}

const page = `<!doctype html>
<html lang="en">
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">
<title>explanations — demo</title>
<link rel="stylesheet" href="/themes/tokens.css">
<link rel="stylesheet" href="/themes/light.css">
<link rel="stylesheet" href="/themes/dark.css">
<style>
  body { font-family: system-ui, sans-serif; margin: 2rem auto; max-width: 46rem; padding: 0 1rem; }
  section { display: block; margin: 2rem 0; }
  #theme-toggle { cursor: pointer; }
</style>
</head>
<body>
<h1>explanations — demo</h1>
<p>One render, two themes: the education explanation starts light, the data explanation starts dark. The toggle swaps the theme classes.</p>
<button id="theme-toggle" type="button">Toggle theme</button>
${section("education", "light", educationIr)}
${section("data", "dark", dataIr)}
<script>
  document.getElementById("theme-toggle").addEventListener("click", () => {
    document.querySelectorAll("section.exp").forEach(el => {
      el.classList.toggle("exp-theme-light")
      el.classList.toggle("exp-theme-dark")
    })
  })
</script>
</body>
</html>`

const server = createServer(async (req, res) => {
  const url = new URL(req.url ?? "/", "http://localhost")
  if (url.pathname === "/") {
    res.writeHead(200, { "content-type": "text/html; charset=utf-8" })
    res.end(page)
    return
  }
  if (url.pathname.startsWith("/themes/")) {
    try {
      const css = await readFile(new URL(`../dist${url.pathname}`, import.meta.url))
      res.writeHead(200, { "content-type": "text/css; charset=utf-8" })
      res.end(css)
    } catch {
      res.writeHead(404, { "content-type": "text/plain" })
      res.end("not found")
    }
    return
  }
  res.writeHead(404, { "content-type": "text/plain" })
  res.end("not found")
})

server.listen(port, () => {
  console.log(`Demo running at http://localhost:${port}`)
})
```
